# Setup partition function and scheme
CREATE PARTITION FUNCTION IntPartitionFunction (int) AS RANGE RIGHT FOR VALUES (0, 500, 1000)
CREATE PARTITION SCHEME IntPartitionScheme AS PARTITION IntPartitionFunction ALL TO ([PRIMARY])

# int partitioned table test
CREATE TABLE sourceTable(c1 int, c2 CHAR(1024))
CREATE TABLE destinationPartitionedTable(c1 int, c2 CHAR(1024)) ON IntPartitionScheme(c1)
INSERT INTO sourceTable VALUES (-100, 'Partition 1 - Negative')
INSERT INTO sourceTable VALUES (250, 'Partition 2 - 0 to 500')
INSERT INTO sourceTable VALUES (750, 'Partition 3 - 500 to 1000')
INSERT INTO sourceTable VALUES (1500, 'Partition 4 - Above 1000')
INSERT INTO sourceTable VALUES (NULL, 'NULL partition value')
bcp#!#out#!#bcp_source#!#sourceTable
bcp#!#in#!#bcp_source#!#destinationPartitionedTable
SELECT * FROM sourceTable ORDER BY c1
SELECT * FROM destinationPartitionedTable ORDER BY c1
DROP TABLE sourceTable
DROP TABLE destinationPartitionedTable

# Large data volume test with partitioned table
CREATE PARTITION FUNCTION LargeDataPartitionFunction (int) AS RANGE RIGHT FOR VALUES (100, 200, 300, 400, 500)
CREATE PARTITION SCHEME LargeDataPartitionScheme AS PARTITION LargeDataPartitionFunction ALL TO ([PRIMARY])
CREATE TABLE sourceTable(c1 int, c2 CHAR(100))
CREATE TABLE destinationPartitionedTable(c1 int, c2 CHAR(100)) ON LargeDataPartitionScheme(c1)
INSERT INTO sourceTable SELECT n, 'Data for partition ' + CAST((CASE 
    WHEN n < 100 THEN 1
    WHEN n < 200 THEN 2
    WHEN n < 300 THEN 3
    WHEN n < 400 THEN 4
    WHEN n < 500 THEN 5
    ELSE 6
END) AS VARCHAR(10)) FROM (SELECT TOP 600 ROW_NUMBER() OVER (ORDER BY (SELECT NULL)) AS n FROM master.dbo.spt_values t1 CROSS JOIN master.dbo.spt_values t2) t
bcp#!#out#!#bcp_source#!#sourceTable
bcp#!#in#!#bcp_source#!#destinationPartitionedTable
SELECT COUNT(*) FROM sourceTable
SELECT COUNT(*) FROM destinationPartitionedTable
SELECT COUNT(*) FROM sourceTable WHERE c1 < 100
SELECT COUNT(*) FROM destinationPartitionedTable WHERE c1 < 100
SELECT COUNT(*) FROM sourceTable WHERE c1 >= 100 AND c1 < 200
SELECT COUNT(*) FROM destinationPartitionedTable WHERE c1 >= 100 AND c1 < 200
DROP TABLE sourceTable
DROP TABLE destinationPartitionedTable
DROP PARTITION SCHEME LargeDataPartitionScheme
DROP PARTITION FUNCTION LargeDataPartitionFunction

# Check constraints with partitioned table
CREATE PARTITION FUNCTION ConstraintPartitionFunction (int) AS RANGE RIGHT FOR VALUES (0, 500, 1000)
CREATE PARTITION SCHEME ConstraintPartitionScheme AS PARTITION ConstraintPartitionFunction ALL TO ([PRIMARY])
CREATE TABLE sourceTable(c1 int, c2 varchar(50))
CREATE TABLE destinationPartitionedTable(c1 int, c2 varchar(50) CHECK (LEN(c2) > 5)) ON ConstraintPartitionScheme(c1)
INSERT INTO sourceTable VALUES (-100, 'Long enough string')
INSERT INTO sourceTable VALUES (250, 'Another valid string')
INSERT INTO sourceTable VALUES (750, 'Valid for constraint')
INSERT INTO sourceTable VALUES (1500, 'This works too')
INSERT INTO sourceTable VALUES (2000, 'Short')
bcp#!#out#!#bcp_source#!#sourceTable
bcp#!#in#!#bcp_source#!#destinationPartitionedTable
SELECT * FROM sourceTable ORDER BY c1
SELECT * FROM destinationPartitionedTable ORDER BY c1
-- Now try with -h "CHECK_CONSTRAINTS" to ignore constraints
bcp#!#out#!#bcp_source#!#sourceTable
bcp -h "CHECK_CONSTRAINTS"#!#in#!#bcp_source#!#destinationPartitionedTable
SELECT * FROM sourceTable ORDER BY c1
SELECT * FROM destinationPartitionedTable ORDER BY c1
DROP TABLE sourceTable
DROP TABLE destinationPartitionedTable
DROP PARTITION SCHEME ConstraintPartitionScheme
DROP PARTITION FUNCTION ConstraintPartitionFunction
