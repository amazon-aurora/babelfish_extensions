# INSERT BULK type validation: SqlBulkCopy per-column coercion and statement-time
# declared-to-target Gate 1 convertibility checks.

# --- SqlBulkCopy per-column coercion (data loads with implicit conversion) ---

# int -> bigint (widening), including NULL
Create table src_int(a int, b int)
Create table dst_bigint(a bigint, b bigint)
Insert into src_int values (12345, 1)
Insert into src_int values (NULL, 2)
insertbulk#!#src_int#!#dst_bigint
Select * from dst_bigint
drop table src_int
drop table dst_bigint

# smallint -> bigint (widening)
Create table src_smallint(a smallint)
Create table dst_bigint2(a bigint)
Insert into src_smallint values (100)
Insert into src_smallint values (-100)
insertbulk#!#src_smallint#!#dst_bigint2
Select * from dst_bigint2
drop table src_smallint
drop table dst_bigint2

# bigint -> smallint (narrowing, in range), two columns
Create table src_bigint3(a bigint, b bigint)
Create table dst_smallint(a smallint, b smallint)
Insert into src_bigint3 values (100, 1)
Insert into src_bigint3 values (-100, 2)
insertbulk#!#src_bigint3#!#dst_smallint
Select * from dst_smallint
drop table src_bigint3
drop table dst_smallint

# varchar(10) -> varchar(50) (widening)
Create table src_vc10(a varchar(10))
Create table dst_vc50(a varchar(50))
Insert into src_vc10 values ('hello')
Insert into src_vc10 values (NULL)
insertbulk#!#src_vc10#!#dst_vc50
Select * from dst_vc50
drop table src_vc10
drop table dst_vc50

# varchar(50) -> varchar(10): a value longer than the target is rejected by bulk load
# (invalid column length), so no row is inserted -- confirms safe handling, no truncation
Create table src_vc50(a varchar(50))
Create table dst_vc10(a varchar(10))
Insert into src_vc50 values ('0123456789ABCDEF')
Insert into src_vc50 values (NULL)
insertbulk#!#src_vc50#!#dst_vc10
Select * from dst_vc10
drop table src_vc50
drop table dst_vc10

# varbinary(10) -> varbinary(50)
Create table src_vb10(a varbinary(10))
Create table dst_vb50(a varbinary(50))
Insert into src_vb10 values (0x0102)
Insert into src_vb10 values (NULL)
insertbulk#!#src_vb10#!#dst_vb50
Select * from dst_vb50
drop table src_vb10
drop table dst_vb50

# int identity -> bigint identity
Create table src_ident(a int identity, b int)
Create table dst_ident(a bigint identity, b bigint)
 set IDENTITY_INSERT src_ident ON
Insert into src_ident (a, b) values (10, 1)
Insert into src_ident (a, b) values (20, 2)
 set IDENTITY_INSERT src_ident OFF
insertbulk#!#src_ident#!#dst_ident
Select * from dst_ident
Select Ident_CURRENT('dst_ident')
drop table src_ident
drop table dst_ident

# --- Gate 1: statement-time declared -> target convertibility (raw INSERT BULK) ---

# Convertible, widening declared (int -> bigint target): accepted; pending bulk then makes the next request an error.
CREATE TABLE gate_widen(a bigint)
INSERT BULK gate_widen(a int)
SELECT 1
DROP TABLE gate_widen

# Convertible, narrowing declared (bigint -> int target): accepted.
CREATE TABLE gate_narrow(a int NOT NULL)
INSERT BULK gate_narrow(a bigint)
SELECT 1
DROP TABLE gate_narrow

# Reordered column list is accepted.
CREATE TABLE gate_reorder(a int, b varchar(50))
INSERT BULK gate_reorder(b varchar(50), a int)
SELECT 1
DROP TABLE gate_reorder

# Length difference is accepted (varchar(50) -> varchar(10)).
CREATE TABLE gate_len(a varchar(10))
INSERT BULK gate_len(a varchar(50))
SELECT 1
DROP TABLE gate_len

# Not convertible (varchar -> varbinary): rejected at statement time; session stays usable.
CREATE TABLE gate_rej1(a varbinary(50))
INSERT BULK gate_rej1(a varchar(50))
SELECT 1
DROP TABLE gate_rej1

# Not convertible (uniqueidentifier -> int): rejected at statement time.
CREATE TABLE gate_rej2(a int)
INSERT BULK gate_rej2(a uniqueidentifier)
SELECT 1
DROP TABLE gate_rej2
