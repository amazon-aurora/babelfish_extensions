# Basic insertbulk test with regular source table and int partitioned destination table
CREATE TABLE sourceTable(c1 int, c2 CHAR(1024))
INSERT INTO sourceTable SELECT generate_series(1, 1000, 1), 'Foo'
CREATE PARTITION FUNCTION IntPartitionFunction (int) AS RANGE RIGHT FOR VALUES (0, 500, 1000)
CREATE PARTITION SCHEME IntPartitionScheme AS PARTITION IntPartitionFunction ALL TO ([PRIMARY])
CREATE TABLE partitionedTable(c1 int, c2 CHAR(1024)) ON IntPartitionScheme(c1)
insertbulk#!#sourceTable#!#partitionedTable
Select count(*) from sourceTable
select count(*) from partitionedTable
select $PARTITION.IntPartitionFunction(c1) as partition_number, count(*) from partitionedTable group by $PARTITION.IntPartitionFunction(c1) order by partition_number
DROP TABLE partitionedTable
DROP PARTITION SCHEME IntPartitionScheme
DROP PARTITION FUNCTION IntPartitionFunction
DROP TABLE sourceTable

insert into sourceTable values (2)
insertbulk#!#sourceTable#!#partitionedTable
Select * from sourceTable
